# Banco (migrations, RLS, seeds) — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) ou superpowers:executing-plans. Steps usam checkbox (`- [ ]`).

**Goal:** Criar no Supabase compartilhado as 16 tabelas novas do Atendimento (prefixo `atendimento_`/`departamentos`/`contatos`), com RLS por departamento, helpers, permissões e seeds, tudo **aditivo** e sem tocar em nada do painel.

**Architecture:** Migrations SQL no repo do painel (`painel-implantacao-v2/supabase/migrations/`), aplicadas via Supabase CLI (`db push`). Três migrations coesas: schema, RLS+permissões, seeds. RLS reusa os helpers do painel (`current_user_id()`, `can()`) e adiciona dois helpers próprios (`atende_departamento`, `e_plantonista`).

**Tech Stack:** PostgreSQL (Supabase), Supabase CLI.

## Global Constraints

- **Só aditivo:** apenas `CREATE TABLE/FUNCTION/POLICY/TRIGGER` e `UPDATE permissoes` (para capacidades). **Nenhum** `ALTER`/`DROP` em tabela existente do painel. Não afetar produção.
- **Local dos arquivos:** `painel-implantacao-v2/supabase/migrations/` (fonte única do schema). Nomes gerados pelo CLI (`npx supabase migration new <slug>`), aplicados por `npx supabase db push`. **Nunca** alterar tabela pelo Studio.
- **Padrões do painel (verbatim):** PK `id UUID PRIMARY KEY DEFAULT gen_random_uuid()`; timestamps `TIMESTAMPTZ NOT NULL DEFAULT NOW()`; helpers `SECURITY DEFINER` + `SET search_path = public` + `GRANT EXECUTE ... TO authenticated`; policies `TO authenticated`, nome `<tabela>_<acao>`, escrita gated por `can('<slug>')` no `USING` e `WITH CHECK`.
- **Ações = slugs em `permissoes.capacidades` (`text[]`)**, NÃO tabela `acoes`. Formato `atendimento.<acao>`. Seed = adicionar slug ao array via `UPDATE`. Manter em sincronia com `src/lib/acoes.ts` (já criado no Plano 1).
- **Idempotência:** usar `CREATE TABLE IF NOT EXISTS`, `CREATE OR REPLACE FUNCTION`, `DROP POLICY IF EXISTS` antes de `CREATE POLICY`, `DROP TRIGGER IF EXISTS` antes de `CREATE TRIGGER`.

  NEW.updated_at = NOW();

```sql
-- Atendimento GR7: helpers de visibilidade + RLS (aditivo).

-- Helper: usuario logado atende o departamento?
CREATE OR REPLACE FUNCTION public.atende_departamento(dep_id UUID)
RETURNS BOOLEAN LANGUAGE SQL STABLE SECURITY DEFINER SET search_path = public AS $$
  SELECT EXISTS (
    SELECT 1 FROM atendente_departamentos ad
    WHERE ad.departamento_id = dep_id
      AND ad.usuario_id = current_user_id()
  );
$$;
GRANT EXECUTE ON FUNCTION public.atende_departamento(UUID) TO authenticated;

-- Helper: usuario logado e plantonista do turno?
CREATE OR REPLACE FUNCTION public.e_plantonista(p_id UUID)
RETURNS BOOLEAN LANGUAGE SQL STABLE SECURITY DEFINER SET search_path = public AS $$
  SELECT EXISTS (
    SELECT 1 FROM atendimento_plantao_usuarios pu
    WHERE pu.plantao_id = p_id
      AND pu.usuario_id = current_user_id()
  );
$$;
GRANT EXECUTE ON FUNCTION public.e_plantonista(UUID) TO authenticated;

-- ===== Catalogos e config: SELECT autenticado; escrita = can('atendimento.config') =====
DO $$
DECLARE t TEXT;
BEGIN
  FOREACH t IN ARRAY ARRAY[
    'departamentos','atendente_departamentos','atendimento_tags','atendimento_motivos',
    'atendimento_mensagens_rapidas','atendimento_plantoes','atendimento_plantao_usuarios',
    'bot_mensagens','atendimento_horarios','atendimento_config'
  ] LOOP
    EXECUTE format('ALTER TABLE public.%I ENABLE ROW LEVEL SECURITY;', t);
    EXECUTE format('DROP POLICY IF EXISTS %I ON public.%I;', t||'_select', t);
    EXECUTE format('CREATE POLICY %I ON public.%I FOR SELECT TO authenticated USING (auth.uid() IS NOT NULL);', t||'_select', t);
    EXECUTE format('DROP POLICY IF EXISTS %I ON public.%I;', t||'_insert', t);
    EXECUTE format('CREATE POLICY %I ON public.%I FOR INSERT TO authenticated WITH CHECK (can(''atendimento.config''));', t||'_insert', t);
    EXECUTE format('DROP POLICY IF EXISTS %I ON public.%I;', t||'_update', t);
    EXECUTE format('CREATE POLICY %I ON public.%I FOR UPDATE TO authenticated USING (can(''atendimento.config'')) WITH CHECK (can(''atendimento.config''));', t||'_update', t);
    EXECUTE format('DROP POLICY IF EXISTS %I ON public.%I;', t||'_delete', t);
    EXECUTE format('CREATE POLICY %I ON public.%I FOR DELETE TO authenticated USING (can(''atendimento.config''));', t||'_delete', t);
  END LOOP;
END $$;

-- ===== contatos: SELECT autenticado; escrita por quem opera atendimento =====
ALTER TABLE public.contatos ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS contatos_select ON public.contatos;
CREATE POLICY contatos_select ON public.contatos FOR SELECT TO authenticated USING (auth.uid() IS NOT NULL);
DROP POLICY IF EXISTS contatos_insert ON public.contatos;
CREATE POLICY contatos_insert ON public.contatos FOR INSERT TO authenticated WITH CHECK (can('atendimento.responder'));
DROP POLICY IF EXISTS contatos_update ON public.contatos;
CREATE POLICY contatos_update ON public.contatos FOR UPDATE TO authenticated USING (can('atendimento.responder')) WITH CHECK (can('atendimento.responder'));

-- ===== atendimentos: visibilidade por departamento/plantao/responsavel =====
ALTER TABLE public.atendimentos ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS atendimentos_select ON public.atendimentos;
CREATE POLICY atendimentos_select ON public.atendimentos FOR SELECT TO authenticated
  USING (
    atende_departamento(departamento_id)
    OR responsavel_id = current_user_id()
    OR e_plantonista(plantao_id)
  );
DROP POLICY IF EXISTS atendimentos_update ON public.atendimentos;
CREATE POLICY atendimentos_update ON public.atendimentos FOR UPDATE TO authenticated
  USING (
    (atende_departamento(departamento_id) OR responsavel_id = current_user_id() OR e_plantonista(plantao_id))
    AND (can('atendimento.assumir') OR can('atendimento.responder') OR can('atendimento.finalizar') OR can('atendimento.transferir'))
  )
  WITH CHECK (
    atende_departamento(departamento_id) OR responsavel_id = current_user_id() OR e_plantonista(plantao_id)
  );
-- INSERT de atendimentos so pela Edge Function (service role, ignora RLS): sem policy de INSERT p/ authenticated.

-- ===== atendimento_mensagens: herda visibilidade do atendimento pai =====
ALTER TABLE public.atendimento_mensagens ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS atendimento_mensagens_select ON public.atendimento_mensagens;
CREATE POLICY atendimento_mensagens_select ON public.atendimento_mensagens FOR SELECT TO authenticated
  USING (EXISTS (SELECT 1 FROM public.atendimentos a WHERE a.id = atendimento_id
    AND (atende_departamento(a.departamento_id) OR a.responsavel_id = current_user_id() OR e_plantonista(a.plantao_id))));
DROP POLICY IF EXISTS atendimento_mensagens_insert ON public.atendimento_mensagens;
CREATE POLICY atendimento_mensagens_insert ON public.atendimento_mensagens FOR INSERT TO authenticated
  WITH CHECK (can('atendimento.responder') AND EXISTS (SELECT 1 FROM public.atendimentos a WHERE a.id = atendimento_id
    AND (atende_departamento(a.departamento_id) OR a.responsavel_id = current_user_id() OR e_plantonista(a.plantao_id))));

-- ===== atendimento_anexos: herda visibilidade da mensagem pai =====
ALTER TABLE public.atendimento_anexos ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS atendimento_anexos_select ON public.atendimento_anexos;
CREATE POLICY atendimento_anexos_select ON public.atendimento_anexos FOR SELECT TO authenticated
  USING (EXISTS (SELECT 1 FROM public.atendimento_mensagens m JOIN public.atendimentos a ON a.id = m.atendimento_id
    WHERE m.id = mensagem_id
      AND (atende_departamento(a.departamento_id) OR a.responsavel_id = current_user_id() OR e_plantonista(a.plantao_id))));
DROP POLICY IF EXISTS atendimento_anexos_insert ON public.atendimento_anexos;
CREATE POLICY atendimento_anexos_insert ON public.atendimento_anexos FOR INSERT TO authenticated
  WITH CHECK (can('atendimento.responder') AND EXISTS (SELECT 1 FROM public.atendimento_mensagens m JOIN public.atendimentos a ON a.id = m.atendimento_id
    WHERE m.id = mensagem_id
      AND (atende_departamento(a.departamento_id) OR a.responsavel_id = current_user_id() OR e_plantonista(a.plantao_id))));

-- ===== atendimento_tag_vinculos: ver se atendimento visivel; marcar = responder =====
ALTER TABLE public.atendimento_tag_vinculos ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS atendimento_tag_vinculos_select ON public.atendimento_tag_vinculos;
CREATE POLICY atendimento_tag_vinculos_select ON public.atendimento_tag_vinculos FOR SELECT TO authenticated
  USING (EXISTS (SELECT 1 FROM public.atendimentos a WHERE a.id = atendimento_id
    AND (atende_departamento(a.departamento_id) OR a.responsavel_id = current_user_id() OR e_plantonista(a.plantao_id))));
DROP POLICY IF EXISTS atendimento_tag_vinculos_insert ON public.atendimento_tag_vinculos;
CREATE POLICY atendimento_tag_vinculos_insert ON public.atendimento_tag_vinculos FOR INSERT TO authenticated
  WITH CHECK (can('atendimento.responder'));
DROP POLICY IF EXISTS atendimento_tag_vinculos_delete ON public.atendimento_tag_vinculos;
CREATE POLICY atendimento_tag_vinculos_delete ON public.atendimento_tag_vinculos FOR DELETE TO authenticated
  USING (can('atendimento.responder'));

-- ===== atendimento_transferencias: ver se atendimento visivel; inserir = transferir =====
ALTER TABLE public.atendimento_transferencias ENABLE ROW LEVEL SECURITY;
DROP POLICY IF EXISTS atendimento_transferencias_select ON public.atendimento_transferencias;
CREATE POLICY atendimento_transferencias_select ON public.atendimento_transferencias FOR SELECT TO authenticated
  USING (EXISTS (SELECT 1 FROM public.atendimentos a WHERE a.id = atendimento_id
    AND (atende_departamento(a.departamento_id) OR a.responsavel_id = current_user_id() OR e_plantonista(a.plantao_id))));
DROP POLICY IF EXISTS atendimento_transferencias_insert ON public.atendimento_transferencias;
CREATE POLICY atendimento_transferencias_insert ON public.atendimento_transferencias FOR INSERT TO authenticated
  WITH CHECK (can('atendimento.transferir'));
```
